    ```go
    package skill

    // SkillDeps provides dependencies that skills need during initialization.
    type SkillDeps struct {
        ProjectDir string       // .serena/ directory for project-scoped data
        GlobalDir  string       // ~/.serena/ directory for global data
        Logger     *slog.Logger
    }

    // Skill is a reusable capability package (D-13).
    type Skill interface {
        Name() string
        Description() string
        Init(deps SkillDeps) error
    }

    // ToolProvider is a Skill that contributes MCP tools (D-14 tool skill).
    type ToolProvider interface {
        Skill
        Tools() []*mcp.ToolDef
    }

    // WorkflowProvider is a Skill that contributes prompts/context (D-14 workflow skill).
    type WorkflowProvider interface {
        Skill
        Prompts() map[string]string
    }
    ```

    **registry.go** -- Per D-11 (Caddy-style init() registration):
    ```go
    package skill

    var (
        globalMu sync.RWMutex
        skills   = make(map[string]Skill)
    )

    func Register(s Skill)                    // called from init() in skill packages
    func Get(name string) (Skill, bool)       // lookup by name
    func All() []Skill                        // all registered skills
    func ToolProviders() []ToolProvider       // filter to ToolProvider implementations
    func WorkflowProviders() []WorkflowProvider // filter to WorkflowProvider implementations
    func InitAll(deps SkillDeps) error        // calls Init on each registered skill
    func Reset()                              // for testing: clear all registrations
    ```

    **registry_test.go:**

    - Register a mock WorkflowProvider skill, verify WorkflowProviders returns it
    - ToolProviders filters correctly (only returns ToolProvider implementations)
    - InitAll calls Init on all skills with deps
    - Reset clears the registry
    - Duplicate registration overwrites silently (last wins)

    ```go
    package skill

    // ContextSpec defines a tool context loaded from YAML (D-12).
    // Controls which skills and tools are active in a given environment.
    type ContextSpec struct {
        Name        string   `yaml:"name"`
        Description string   `yaml:"description"`
        Skills      []string `yaml:"skills"`       // skill names to activate
        Tools       []string `yaml:"tools"`         // individual tool names to include
        ExcludeTools []string `yaml:"exclude_tools"` // tools to exclude from active skills
    }

    // ModeSpec defines an operational mode loaded from YAML (D-12).
    // Controls behavior patterns within a context.
    type ModeSpec struct {
        Name        string            `yaml:"name"`
        Description string            `yaml:"description"`
        Skills      []string          `yaml:"skills"`
        Tools       []string          `yaml:"tools"`
        ExcludeTools []string         `yaml:"exclude_tools"`
        Prompts     map[string]string `yaml:"prompts"`  // prompt overrides per tool
    }

    // LoadContextSpecs loads context definitions from a YAML file.
    func LoadContextSpecs(path string) ([]ContextSpec, error)

    // LoadModeSpecs loads mode definitions from a YAML file.
    func LoadModeSpecs(path string) ([]ModeSpec, error)

    // ResolveTools takes a spec (context or mode) and the global registry,
    // returns the list of ToolDefs that should be active.
    func ResolveTools(skillNames []string, includeTools []string, excludeTools []string) []*mcp.ToolDef
    ```
