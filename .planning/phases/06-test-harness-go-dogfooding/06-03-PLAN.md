This test validates the full HTTP serialization path per D-03/D-04. It creates an `httptest.NewServer` wrapping the daemon's HTTP handler, then connects an MCP client via `StreamableClientTransport`.

**Important:** The TestDaemon struct's `daemon` field is private. The harness needs to expose a method for HTTP session creation. Two options:
- Option A: Add a `NewHTTPSession(t *testing.T) *mcp.ClientSession` method to TestDaemon in harness.go
- Option B: Export the daemon field or add a `HTTPHandler() http.Handler` accessor to TestDaemon

**Preferred: Option A** -- Add to harness.go (this task should modify harness.go to add the method):

```go
// NewHTTPSession creates an MCP client session over HTTP transport.
// Uses httptest.NewServer + StreamableClientTransport per D-03/D-04.
func (td *TestDaemon) NewHTTPSession(t *testing.T) *mcp.ClientSession {
    t.Helper()
    ts := httptest.NewServer(td.daemon.MCPServer().HTTPHandler())
    t.Cleanup(ts.Close)
    
    client := mcp.NewClient(&mcp.Implementation{Name: "test-http", Version: "1.0"}, nil)
    session, err := client.Connect(context.Background(), &mcp.StreamableClientTransport{
        Endpoint: ts.URL + "/mcp",
    }, nil)
    require.NoError(t, err)
    t.Cleanup(func() { session.Close() })
    return session
}
```

func TestHTTPSmoke_ToolCallRoundTrip(t *testing.T) {
    td := StartTestDaemon(t, Options{SkipLS: true})
    httpSession := td.NewHTTPSession(t)
    
    // Test 1: list_memories via HTTP -- proves full JSON serialization path
    result, err := httpSession.CallTool(context.Background(), &mcp.CallToolParams{
        Name:      "list_memories",
        Arguments: map[string]any{},
    })
    require.NoError(t, err)
    require.False(t, result.IsError)
    // Result may be empty list but call must succeed
    
    // Test 2: write + read via HTTP -- proves bidirectional data integrity
    _, err = httpSession.CallTool(context.Background(), &mcp.CallToolParams{
        Name:      "write_memory",
        Arguments: map[string]any{"name": "http-test", "content": "via HTTP transport"},
    })
    require.NoError(t, err)
    
    readResult, err := httpSession.CallTool(context.Background(), &mcp.CallToolParams{
        Name:      "read_memory",
        Arguments: map[string]any{"name": "http-test"},
    })
    require.NoError(t, err)
    assert.Contains(t, textContent(readResult), "via HTTP transport")
}
