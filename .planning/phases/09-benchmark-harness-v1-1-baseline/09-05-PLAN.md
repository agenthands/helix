       // heapSnapshot writes a gzipped pprof heap profile to test/bench/pprof/{scenario}-{sha}.pb.gz
       // per D-07/D-08. Calls runtime.GC() first per Pattern 5 to stabilize short-lived allocs.
       // Pitfall 8: files are git-ignored; committed baseline snapshots are a release-time concern.
       func heapSnapshot(tb testing.TB, scenario string) {
           tb.Helper()
           runtime.GC()
           sha := gitShortSHA(tb)
           dir := filepath.Join("pprof")
           if err := os.MkdirAll(dir, 0o755); err != nil {
               tb.Fatalf("mkdir pprof: %v", err)
           }
           path := filepath.Join(dir, fmt.Sprintf("%s-%s.pb.gz", scenario, sha))
           f, err := os.Create(path)
           if err != nil {
               tb.Fatalf("create heap snapshot: %v", err)
           }
           defer f.Close()
           if err := pprof.WriteHeapProfile(f); err != nil {
               tb.Fatalf("write heap profile: %v", err)
           }
           tb.Logf("heap snapshot: %s", path)
       }
       ```

       func reportMemory(b *testing.B, label string) {
           b.Helper()
           var ms runtime.MemStats
           runtime.ReadMemStats(&ms)
           kernelRSS, err := rss.CurrentRSS()
           if err != nil {
               b.Logf("%s: rss_go_sys_bytes=%d rss_kernel_bytes=unavailable (%v)", label, ms.Sys, err)
           } else {
               b.Logf("%s: rss_go_sys_bytes=%d rss_kernel_bytes=%d", label, ms.Sys, kernelRSS)
           }
           // Also surface as benchmark custom metrics so benchstat picks them up.
           b.ReportMetric(float64(ms.Sys), label+"_go_sys_B")
           if err == nil {
               b.ReportMetric(float64(kernelRSS), label+"_kernel_B")
           }
       }
       ```
    3. Implement `BenchmarkMemory` as a single bench that runs exactly once (use `-benchtime=1x` in the verify command) and walks the 4 scenarios in order:
       ```go
       func BenchmarkMemory(b *testing.B) {
           requireGoplsB(b)
           b.ReportAllocs()
           for b.Loop() {
               // Scenario 1: daemon idle (just started, no workspace)
               bd := startBenchDaemon(b)
               reportMemory(b, "s1_idle")
               heapSnapshot(b, "s1-idle")

               // Scenario 2: post-workspace-activation (LS running, indexed)
               fixture := prepareGoFixtureB(b)
               activateWorkspaceB(b, bd, fixture)
               reportMemory(b, "s2_workspace")
               heapSnapshot(b, "s2-workspace")

               // Scenario 3: post-first-tool-call (LS worker warm)
               _ = callToolB(b, bd.Session, "find_symbol", benchToolArgsFor("find_symbol"))
               reportMemory(b, "s3_first_call")
               heapSnapshot(b, "s3-first-call")

               // Scenario 4: post-100-calls (drift detection)
               for i := 0; i < 100; i++ {
                   _ = callToolB(b, bd.Session, "find_symbol", benchToolArgsFor("find_symbol"))
               }
               reportMemory(b, "s4_post_100_calls")
               heapSnapshot(b, "s4-post-100-calls")

               bd.Stop()
           }
       }

       // benchToolArgsFor returns the args from tools_manifest for a given tool name,
       // or fatals if missing. This keeps the memory bench in lockstep with the manifest
       // so args can never drift.
       func benchToolArgsFor(name string) map[string]any {
           for _, tc := range benchTools {
               if tc.name == name {
                   return tc.args
               }
           }
           panic("benchToolArgsFor: tool not in manifest: " + name)
       }
       ```
