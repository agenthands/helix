    File contents outline:

    ```go
    // Package bench_test: Phase 10 slog hot-path allocation benchmark.
    // Proves OBS-06: obs.ContextHandler adds ≤ +1 alloc/op vs the unwrapped baseline.
    package bench_test

    import (
        "context"
        "io"
        "log/slog"
        "testing"

        "github.com/postfix/serena/internal/obs"
    )

    // newDiscardJSONHandler returns the same base handler shape the daemon installs
    // in cli/root.go runDaemon — JSON to io.Discard for zero IO noise in the bench.
    func newDiscardJSONHandler() slog.Handler {
        return slog.NewJSONHandler(io.Discard, &slog.HandlerOptions{Level: slog.LevelInfo})
    }

    // BenchmarkSlogHotPath_Baseline measures a bare slog.NewJSONHandler with no wrapping.
    // This is the Phase 9 reference shape.
    func BenchmarkSlogHotPath_Baseline(b *testing.B) {
        logger := slog.New(newDiscardJSONHandler())
        ctx := context.Background()
        b.ReportAllocs()
        b.ResetTimer()
        for b.Loop() {
            logger.LogAttrs(ctx, slog.LevelInfo, "tool_call",
                slog.String("tool", "find_symbol"),
                slog.String("workspace", "bench"),
                slog.Int("latency_ms", 42),
            )
        }
    }

    // BenchmarkSlogHotPath_WithContextHandler measures the same emission path
    // through obs.NewContextHandler. Phase 10 stub always takes the fast path,
    // so the allocation delta must be ≤ +1 alloc/op (OBS-06 contract).
    func BenchmarkSlogHotPath_WithContextHandler(b *testing.B) {
        logger := slog.New(obs.NewContextHandler(newDiscardJSONHandler()))
        ctx := context.Background()
        b.ReportAllocs()
        b.ResetTimer()
        for b.Loop() {
            logger.LogAttrs(ctx, slog.LevelInfo, "tool_call",
                slog.String("tool", "find_symbol"),
                slog.String("workspace", "bench"),
                slog.Int("latency_ms", 42),
            )
        }
    }
    ```

    Constraints:
    - Use `b.Loop()` (Go 1.25 idiom) not `for i := 0; i < b.N; i++` — matches Phase 9 BENCH-01.
    - `b.ReportAllocs()` MUST be called before the loop.
    - Use `slog.LogAttrs` with typed attrs — NEVER `slog.Info("msg", "key", val)` which boxes (Pitfall #3 from canonical refs).
    - Emit to io.Discard so IO cost is zero and allocations reflect only slog + wrapper work.
    - Do NOT introduce any new module dependency — stdlib + internal/obs only.
    - Do NOT modify any existing bench files — strictly additive.

    After creating the file, run the two benchmarks and capture the output for Task 2:
    ```
    go test ./test/bench/ -run=^$ -bench='BenchmarkSlogHotPath' -benchmem -count=6
    ```
    Save the output text verbatim; Task 2 writes it into the baseline file.
