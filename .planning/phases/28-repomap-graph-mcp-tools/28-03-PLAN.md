**ensureGraph():**
```go
func (s *RepoMapSkill) ensureGraph() error {
    s.mu.Lock()
    defer s.mu.Unlock()
    
    ver := s.cache.Version()
    if s.graph != nil && s.graphVer == ver {
        return nil // Graph is current
    }
    
    graph, err := repomap.BuildGraph(s.cache)
    if err != nil {
        return serr.Wrap(serr.Internal, "building reference graph", err)
    }
    s.graph = graph
    s.graphVer = ver
    return nil
}
```
