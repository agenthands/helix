    - If State == "degraded": `p.Warning("%s (%s) degraded", w.Language, w.Command)` -- use yellow color
    - If State == "failed": `p.Failure("%s (%s) failed", w.Language, w.Command)`
  - For each circuit in ws.Circuits with State != "closed":
    - `p.Failure("%s circuit breaker %s (%d failures)", c.Language, c.State, c.Failures)`

`func runStatus(cmd *cobra.Command, args []string) error`:

1. Read flags: `jsonOut, _ := cmd.Flags().GetBool("json")` and `verbose, _ := cmd.Flags().GetBool("verbose")`
2. Determine socket path: `socketPath := config.DefaultSocketPath()`
3. Check if daemon is running per D-07:
   ```go
   if _, err := os.Stat(socketPath); os.IsNotExist(err) {
       fmt.Fprintln(os.Stderr, "Daemon not running")
       os.Exit(1)
   }
   // Also try a quick TCP dial to verify socket is alive
   testConn, err := net.DialTimeout("unix", socketPath, 2*time.Second)
   if err != nil {
       fmt.Fprintln(os.Stderr, "Daemon not running")
       os.Exit(1)
   }
   testConn.Close()
   ```
4. Connect via gRPC (simplified version of forwarder dial -- no auto-start, no otel):
   ```go
   conn, err := grpc.NewClient("unix://"+socketPath,
       grpc.WithTransportCredentials(insecure.NewCredentials()),
   )
   if err != nil {
       return fmt.Errorf("connecting to daemon: %w", err)
   }
   defer conn.Close()
   client := serenav1.NewForwarderServiceClient(conn)
   ```
5. Call GetStatus:
   ```go
   ctx, cancel := context.WithTimeout(cmd.Context(), 5*time.Second)
   defer cancel()
   resp, err := client.GetStatus(ctx, &serenav1.StatusRequest{Verbose: verbose})
   if err != nil {
       return fmt.Errorf("querying daemon status: %w", err)
   }
   ```
6. Unmarshal response payload:
   ```go
   var report lspool.HealthReport
   if err := json.Unmarshal(resp.Payload, &report); err != nil {
       return fmt.Errorf("parsing health report: %w", err)
   }
   ```
7. If `--json` per D-08: always use verbose data (re-query with verbose=true if needed, or just pretty-print what we got). If `jsonOut`:
   ```go
   if jsonOut {
       // Re-query with verbose=true for full data per D-08
       if !verbose {
           resp, err = client.GetStatus(ctx, &serenav1.StatusRequest{Verbose: true})
           if err != nil { return err }
           if err := json.Unmarshal(resp.Payload, &report); err != nil { return err }
       }
       out, _ := json.MarshalIndent(report, "", "  ")
       fmt.Println(string(out))
   }
   ```
   Otherwise, use StatusPrinter:
   ```go
   printer := &StatusPrinter{}
   printer.PrintReport(&report, verbose)
   ```
8. Exit code 1 if any failures per D-07:
   ```go
   printer := &StatusPrinter{}
   if printer.HasFailures(&report) {
       os.Exit(1)
   }
   ```
