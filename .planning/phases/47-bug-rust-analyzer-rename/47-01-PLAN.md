3. Add method `ensureReadyCh` and `signalReady` (private helpers):
```go
func (r *RustAnalyzerAdapter) ensureReadyCh() chan struct{} {
    r.readyChMu.Lock()
    defer r.readyChMu.Unlock()
    if r.readyCh == nil {
        r.readyCh = make(chan struct{})
    }
    return r.readyCh
}

func (r *RustAnalyzerAdapter) signalReady() {
    r.readyChMu.Lock()
    defer r.readyChMu.Unlock()
    if r.readyCh != nil {
        select {
        case <-r.readyCh:
            // already closed
        default:
            close(r.readyCh)
        }
    } else {
        ch := make(chan struct{})
        close(ch)
        r.readyCh = ch
    }
}

func (r *RustAnalyzerAdapter) resetReadyCh() {
    r.readyChMu.Lock()
    defer r.readyChMu.Unlock()
    r.readyCh = make(chan struct{})
}
```

5. Implement `WaitUntilRenameReady` as an exported method on `*RustAnalyzerAdapter`:
```go
// WaitUntilRenameReady blocks until rust-analyzer reports quiescent=true via
// experimental/serverStatus, ctx is cancelled, or renameReadinessTimeout
// elapses. Returns true iff the server reached quiescence before the
// timeout / cancellation. Safe to call concurrently.
func (r *RustAnalyzerAdapter) WaitUntilRenameReady(ctx context.Context) bool {
    if r.quiescent.Load() {
        return true
    }
    ch := r.ensureReadyCh()
    deadline := time.NewTimer(renameReadinessTimeout)
    defer deadline.Stop()
    select {
    case <-ch:
        return r.quiescent.Load()
    case <-ctx.Done():
        return false
    case <-deadline.C:
        return false
    }
}
```
