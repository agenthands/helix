    ```go
    package upgrade

    import (
        "strings"

        "golang.org/x/mod/semver"
    )

    // Canonical normalizes "1.9.0" or "v1.9.0" → "v1.9.0" (semver requires the leading 'v').
    func Canonical(v string) string {
        if !strings.HasPrefix(v, "v") {
            return "v" + v
        }
        return v
    }

    // IsDowngrade returns true if target <= current per semver.Compare.
    // Equal versions are treated as downgrade per D-10 ("already up to date").
    func IsDowngrade(current, target string) bool {
        return semver.Compare(Canonical(target), Canonical(current)) <= 0
    }

    // IsPrerelease reports whether tag has a SemVer pre-release segment (e.g., "-rc1").
    func IsPrerelease(tag string) bool {
        return semver.Prerelease(Canonical(tag)) != ""
    }
    ```

    ```go
    package upgrade

    import (
        "fmt"
        "os"
        "path/filepath"
        "strings"
    )

    // ProbeWritable returns nil iff the directory containing installPath is writable
    // by the current user. Uses os.CreateTemp probe to avoid stat-time TOCTOU.
    func ProbeWritable(installPath string) error {
        dir := filepath.Dir(installPath)
        f, err := os.CreateTemp(dir, ".helix-perm-probe-*")
        if err != nil {
            return fmt.Errorf("install path %s not writable by current user: %w", installPath, err)
        }
        name := f.Name()
        _ = f.Close()
        _ = os.Remove(name)
        return nil
    }

    // SudoHint formats the actionable re-invocation per D-09.
    func SudoHint(installPath string, args []string) string {
        return fmt.Sprintf(
            "helix is installed at %s and your user cannot write there.\nRe-run with elevated privileges, e.g.:\n  sudo helix upgrade %s",
            installPath, strings.Join(args, " "),
        )
    }

    package upgrade

    import (
        "fmt"

        "github.com/jedisct1/go-minisign"
    )

    // VerifyArchive verifies sigPath is a valid minisign signature for archivePath
    // using the embedded pubKeyBytes. Returns a SINGLE canonical error on any failure
    // (tampered, wrong-key, missing) per Pitfall 4 — attackers must not be able to
    // distinguish failure modes by error message.
    //
    // testPubKeyOverride allows tests to swap the embedded pubkey for testdata/test_keypair.pub.
    var testPubKeyOverride []byte

    func currentPubKey() []byte {
        if testPubKeyOverride != nil {
            return testPubKeyOverride
        }
        return pubKeyBytes
    }

    func VerifyArchive(archivePath, sigPath string) error {
        pub, err := minisign.DecodePublicKey(string(currentPubKey()))
        if err != nil {
            return fmt.Errorf("signature verification FAILED")
        }
        sig, err := minisign.NewSignatureFromFile(sigPath)
        if err != nil {
            return fmt.Errorf("signature verification FAILED")
        }
        ok, err := pub.VerifyFromFile(archivePath, sig)
        if err != nil || !ok {
            return fmt.Errorf("signature verification FAILED")
        }
        return nil
    }
    ```

    func swap(currentPath, newPath string) error {
        oldPath := currentPath + ".old"

        if err := os.Rename(currentPath, oldPath); err != nil {
            return fmt.Errorf("renaming current to .old: %w", err)
        }

            _ = os.Rename(oldPath, currentPath)

        }
        return nil
    }

    func relaunch(binPath string, args, env []string) error {
        cmd := exec.Command(binPath, args[1:]...)
        cmd.Stdin, cmd.Stdout, cmd.Stderr = os.Stdin, os.Stdout, os.Stderr
        cmd.Env = env
        if err := cmd.Start(); err != nil {
            return err
        }
        os.Exit(0)
        return nil
    }

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Network → user binary | `helix upgrade` fetches release archives over HTTPS from GitHub; the embedded minisign pubkey is the trust anchor — any path that bypasses verification produces an unsigned-binary install. |
| Filesystem → running process | The atomic swap on Unix relies on inode-replace semantics; cross-filesystem rename falls back to non-atomic copy with a visible half-replaced-binary intermediate state. |
| Running process → child process | Daemon detection uses an env var; missing the var setter in daemon spawn means an in-daemon `helix upgrade` would mistakenly try to swap the running binary, breaking the daemon's child processes. |
| Embedded pubkey → upgrade-time verification | The build-time-synced `internal/upgrade/minisign.pub` (Plan 01) is the verification root; any drift between it and the repo-root key produces a verifier that cannot validate signatures from the canonical signing flow. |
| OS env → process | `GITHUB_TOKEN` (if set) authenticates API calls; an attacker controlling the env can spoof the rate-limit identity, but signature verification downstream catches any malicious release content. |
