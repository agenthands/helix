    func TestMetrics_CardinalityBounds(t *testing.T) {

        languages := make([]string, 0, 52)
        for i := 0; i < 52; i++ {
            languages = append(languages, "lang" + string(rune('a' + (i % 26))) + string(rune('a' + (i / 26))))
