**Testdata expected.json — order of operations (W9):** every per-language task follows this sequence:
  1. Implement `provider.go` + `queries.scm` against Tasks 1's testutil helpers; ensure `tree_sitter.NewQuery` compiles cleanly.
  2. Author `before.<ext>` files for every scenario directory (and `after.<ext>` where the scenario asserts a transition); leave `expected.json` absent.
  3. Run `go test ./internal/semantic/extract/<lang>/... -update -count=1` LOCALLY to generate `expected.json` from the provider's actual emit.
  4. MANUALLY inspect a representative sample (at minimum: `function_basic`, `class_or_struct_basic`, `import_named`, one stable-ID transition) and confirm the captures are sane (no missing symbols, no spurious refs, sorted, relative paths). If any sample is wrong, fix the provider/queries.scm and re-run `-update`; iterate until samples are correct.
  5. Run `go test ./internal/semantic/extract/<lang>/... -count=1` (NO -update) twice consecutively; both runs must pass with byte-identical output (proves determinism).
  6. Commit ALL `expected.json` files alongside `provider.go` + `queries.scm` in the same atomic commit.

**CI never runs with `-update`.** The flag is dev-only. CI's lack of `-update` is the regression net — any provider drift fails the golden tests on the next CI run.

    package tsextract

    import (
        _ "embed"

        tree_sitter "github.com/tree-sitter/go-tree-sitter"

        "github.com/agenthands/helix/internal/semantic/extract"
        "github.com/agenthands/helix/internal/treesitter"
    )

    //go:embed queries.scm
    var tsQueries string

    type Provider struct {
        // TS+JS overlap is large enough for one provider per CONTEXT.md
        // discretion. We compile against the typescript grammar; the .js/.jsx
        // extensions are claimed under the same provider so .js files run
        // through the typescript parser (which accepts JS as a strict subset
        // syntactically — TS-only constructs simply won't match in .js files).
        // If the planner determines tsx requires a separate grammar, this
        // struct grows a tsxGrammar field and ExtractFile dispatches by ext.
        tsGrammar  *tree_sitter.Language
        tsxGrammar *tree_sitter.Language
        tsQuery    *tree_sitter.Query
        tsxQuery   *tree_sitter.Query
    }

    func NewProvider(grammars *treesitter.GrammarRegistry) extract.Provider {
        ts, ok := grammars.GetLanguage("typescript")
        if !ok {
            panic("tsextract: 'typescript' grammar missing from injected GrammarRegistry")
        }
        tsx, ok := grammars.GetLanguage("tsx")
        if !ok {
            panic("tsextract: 'tsx' grammar missing from injected GrammarRegistry")
        }
        tsq, err := tree_sitter.NewQuery(ts, tsQueries)
        if err != nil {
            panic("tsextract: ts query compile failed: " + err.Message)
        }
        tsxq, err := tree_sitter.NewQuery(tsx, tsQueries)
        if err != nil {
            panic("tsextract: tsx query compile failed: " + err.Message)
        }
        return &Provider{tsGrammar: ts, tsxGrammar: tsx, tsQuery: tsq, tsxQuery: tsxq}
    }

    func (p *Provider) Language() string { return "typescript" }
    func (p *Provider) Extensions() []string {
        return []string{".ts", ".tsx", ".js", ".jsx", ".mjs", ".cjs"}
    }
    func (p *Provider) TreeSitterLanguage() *tree_sitter.Language { return p.tsGrammar }
    func (p *Provider) Queries() string                            { return tsQueries }
    func (p *Provider) SupportsLSPEnrichment() bool                { return true }
    ```

    package extract

    // PartialExtract builds an ExtractedFile representing a partial or
    // unsupported extraction outcome. Used by:
    //   - The scheduler when classifying a file's language returns "other"
    //     (D-05 unsupported_language path — no symbols/refs/imports written).
    //   - The provider's per-file pipeline when parse_error / query_error /
    //     timeout / file_too_large / binary_or_generated / permission_denied
    //     fires (D-05 first-class-with-errors path — partial facts may still
    //     be written by the caller).
    func PartialExtract(file SourceFile, reason PartialReason, err error) *ExtractedFile {
        msg := ""
        if err != nil {
            msg = err.Error()
        }
        status := ExtractionStatusPartial
        if reason == PartialReasonUnsupportedLanguage || reason == PartialReasonFileTooLarge ||
            reason == PartialReasonBinaryOrGenerated {
            status = ExtractionStatusUnsupported
        }
        return &ExtractedFile{
            File: FileFact{
                Path:              file.Path,
                Language:          file.Language,
                ExtractionStatus:  status,
                ExtractionPartial: true,
                PartialReason:     reason,
                ErrorMessage:      msg,
            },
            Partial:       true,
            PartialReason: string(reason),
        }
    }
    ```
