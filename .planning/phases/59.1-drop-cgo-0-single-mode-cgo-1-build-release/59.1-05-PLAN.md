    ```makefile
    release-smoke: ## Smoke-test the linux-amd64 release archive: extract, run daemon, hit MCP tools/list (D-04)
    	@test -f dist/helix_v*_linux_amd64.tar.gz || { \
    	  echo "dist/helix_v*_linux_amd64.tar.gz not found; run \`make release-snapshot\` first"; exit 1; }
    	@SMOKE_DIR="$$(mktemp -d /tmp/helix-release-smoke.XXXXXX)" ; \
    	tar -xzf dist/helix_v*_linux_amd64.tar.gz -C "$$SMOKE_DIR" ; \
    	HELIX_BIN="$$SMOKE_DIR/$$(ls "$$SMOKE_DIR" | head -1)/helix" ; \
    	test -x "$$HELIX_BIN" || { echo "release-smoke: extracted helix binary not executable: $$HELIX_BIN"; rm -rf "$$SMOKE_DIR"; exit 1; } ; \
    	"$$HELIX_BIN" daemon --http-addr 127.0.0.1:17777 --workspace ./testdata/fixtures/go/ > "$$SMOKE_DIR/daemon.log" 2>&1 & \
    	DAEMON_PID=$$! ; \
    	sleep 3 ; \
    	if ! kill -0 "$$DAEMON_PID" 2>/dev/null ; then \
    	  echo "release-smoke: daemon exited early; log:" ; cat "$$SMOKE_DIR/daemon.log" ; rm -rf "$$SMOKE_DIR" ; exit 1 ; \
    	fi ; \
    	TOOL_COUNT=$$(curl -sS -X POST http://127.0.0.1:17777/mcp -H 'Content-Type: application/json' -d '{"jsonrpc":"2.0","id":1,"method":"tools/list","params":{}}' | jq -r '.result.tools | length' 2>/dev/null || echo 0) ; \
    	GRAMMAR_OK=0 ; \
    	grep -qE '23 tree-sitter grammars? registered|registered 23 grammars' "$$SMOKE_DIR/daemon.log" && GRAMMAR_OK=1 ; \
    	kill "$$DAEMON_PID" 2>/dev/null ; wait "$$DAEMON_PID" 2>/dev/null ; \
    	rm -rf "$$SMOKE_DIR" ; \
    	if [ "$$TOOL_COUNT" -lt 41 ]; then echo "release-smoke FAIL: tool count $$TOOL_COUNT < 41"; exit 1; fi ; \
    	if [ "$$GRAMMAR_OK" != 1 ]; then echo "release-smoke FAIL: 23 tree-sitter grammar log line not found"; exit 1; fi ; \
    	echo "release-smoke OK: tools=$$TOOL_COUNT grammars=23"
    ```
