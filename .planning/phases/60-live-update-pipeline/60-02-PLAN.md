    ```go
    package store

    import (
        "context"
        "database/sql"
        "fmt"
        "sync"

        "github.com/agenthands/helix/internal/semantic"
    )

    // OverlayTx is the per-tx interface returned by BeginOverlayTx. Each tx
    // carries a freshly-allocated, monotone, per-workspace write_epoch (D-04)
    // that is stamped onto every fact row written through this handle.
    type OverlayTx struct {
        tx     *sql.Tx
        repoID string
        epoch  uint64
        unlock func()
    }

    // Epoch returns the write_epoch stamped on every fact row written through
    // this tx. Callers may inspect it for logging / metrics; the writer
    // methods stamp it automatically.
    func (t *OverlayTx) Epoch() uint64 { return t.epoch }

    // BeginOverlayTx opens an overlay write transaction for repoID. Allocates
    // a fresh monotone write_epoch under the per-workspace lock. Cross-
    // workspace BeginOverlayTx calls do not serialize (D-04 invariant).
    //
    // The returned OverlayTx MUST be terminated via Commit() or Rollback().
    // Rollback does NOT rewind the epoch (D-04 invariant: "if the tx rolls
    // back, that epoch is just unused").
    func (s *Store) BeginOverlayTx(ctx context.Context, repoID string) (*OverlayTx, error) {
        if repoID == "" {
            return nil, fmt.Errorf("BeginOverlayTx: empty repoID")
        }
        // Acquire per-workspace lock.
        s.overlayLocksMu.Lock()
        mu, ok := s.overlayLocks[repoID]
        if !ok {
            mu = &sync.Mutex{}
            s.overlayLocks[repoID] = mu
        }
        s.overlayLocksMu.Unlock()
        mu.Lock()

        // Open tx.
        tx, err := s.db.BeginTx(ctx, nil)
        if err != nil {
            mu.Unlock()
            return nil, fmt.Errorf("BeginOverlayTx: open tx: %w", err)
        }

        // Ensure a meta row exists for this repoID. INSERT-on-conflict-do-nothing
        // pattern (DuckDB supports ON CONFLICT DO NOTHING on PRIMARY KEY).
        if _, err := tx.ExecContext(ctx,
            `INSERT INTO semantic_live_overlay_meta (repo_id, current_epoch) VALUES (?, 0)
             ON CONFLICT (repo_id) DO NOTHING`, repoID); err != nil {
            _ = tx.Rollback()
            mu.Unlock()
            return nil, fmt.Errorf("BeginOverlayTx: ensure meta row: %w", err)
        }

        // Atomically increment + read back current_epoch.
        if _, err := tx.ExecContext(ctx,
            `UPDATE semantic_live_overlay_meta
                SET current_epoch = current_epoch + 1
              WHERE repo_id = ?`, repoID); err != nil {
            _ = tx.Rollback()
            mu.Unlock()
            return nil, fmt.Errorf("BeginOverlayTx: bump epoch: %w", err)
        }
        var epoch uint64
        if err := tx.QueryRowContext(ctx,
            `SELECT current_epoch FROM semantic_live_overlay_meta WHERE repo_id = ?`,
            repoID).Scan(&epoch); err != nil {
            _ = tx.Rollback()
            mu.Unlock()
            return nil, fmt.Errorf("BeginOverlayTx: read epoch: %w", err)
        }

        return &OverlayTx{tx: tx, repoID: repoID, epoch: epoch, unlock: mu.Unlock}, nil
    }

    // Commit finalizes the tx and releases the per-workspace lock.
    func (t *OverlayTx) Commit() error {
        defer t.unlock()
        return t.tx.Commit()
    }

    // Rollback aborts the tx and releases the per-workspace lock. The
    // current_epoch increment that opened the tx remains committed (D-04:
    // monotone, never rolled back; the epoch slot is just unused).
    func (t *OverlayTx) Rollback() error {
        defer t.unlock()
        return t.tx.Rollback()
    }

    // FlushOverlay is a no-op today; reserved for future cooperative drain
    // semantics if Phase 63's compaction pipeline needs them. Always returns
    // nil. Callers may invoke at shutdown without harm.
    func (s *Store) FlushOverlay(ctx context.Context) error { return nil }
    ```

    ```go
    // the watcher/handler may invoke them when classifier outputs (Phase 60

    func (t *OverlayTx) MarkFileDeleted(ctx context.Context, path string) error {
        _, err := t.tx.ExecContext(ctx,
            `INSERT INTO semantic_live_overlay_files (repo_id, path, file_id, content_hash, language, status, updated_at, write_epoch)
             VALUES (?, ?, 0, '', '', 'deleted', now(), ?)
             ON CONFLICT (repo_id, path) DO UPDATE SET
                 status      = 'deleted',
                 write_epoch = excluded.write_epoch,
                 updated_at  = now()`,

    // MarkSymbolsDeleted tombstones every overlay-symbol row whose file_id
    // matches the supplied list. The classifier (P04) supplies file_ids
    // resolved against semantic_files; on a bare path-deletion the handler
    // resolves the live or base file_id before invoking this helper.
    func (t *OverlayTx) MarkSymbolsDeleted(ctx context.Context, fileIDs []uint64) error {
        if len(fileIDs) == 0 { return nil }
        // Use a parameterized IN-list construction; DuckDB supports ANY-array
        // params via duckdb-go list binding. Fallback: iterate per-id.
        for _, id := range fileIDs {
            if _, err := t.tx.ExecContext(ctx,

                    SET status='deleted', write_epoch=?, updated_at=now()
                  WHERE repo_id=? AND file_id=?`,
                t.epoch, t.repoID, id); err != nil {
                return err
            }
        }
        return nil
    }

    // MarkReferencesDeleted is the references-table analog of MarkSymbolsDeleted.
    func (t *OverlayTx) MarkReferencesDeleted(ctx context.Context, fileIDs []uint64) error {

                    SET status='deleted', write_epoch=?, updated_at=now()
                  WHERE repo_id=? AND file_id=?`,

            }
        }
        return nil
    }

    // MarkEdgesDeleted tombstones every overlay-edge row whose src_node_id

                return err
            }
        }

    }
    ```
