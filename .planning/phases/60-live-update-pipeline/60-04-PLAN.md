```go
package coalescer

import (
    "sort"

    "github.com/agenthands/helix/internal/semantic/live"
)

// CoalesceEvents implements SPEC §16.2 verbatim. Pure: same input slice +
// same threshold yields byte-identical output.
//
// Rules (from SPEC §16.2):
//   modified+modified → modified
//   created+modified  → created
//   created+deleted   → no-op (drop)
//   modified+deleted  → deleted
//   deleted+created   → modified (resurrection)
//   rename(old→new)+modified(new) → rename
//   any other combo  → last-write-wins
//
// Threshold collapse: if len(merged) > threshold, return a single
// {Kind: ChangeBulkUpdate} event tagged with the first observed RepoID.
func CoalesceEvents(events []live.SourceChangeEvent, threshold int) []live.SourceChangeEvent {
    byPath := map[string]live.SourceChangeEvent{}
    for _, ev := range events {
        key := ev.Path
        if ev.Kind == live.ChangeFileRenamed {
            key = ev.OldPath + "→" + ev.Path
        }
        prev, ok := byPath[key]
        if !ok {
            byPath[key] = ev
            continue
        }
        merged, keep := MergeChange(prev, ev)
        if !keep {
            delete(byPath, key)
            continue
        }
        byPath[key] = merged
    }
    result := make([]live.SourceChangeEvent, 0, len(byPath))
    for _, v := range byPath {
        result = append(result, v)
    }
    if threshold > 0 && len(result) > threshold {
        repo := semanticRepoID(events)
        return []live.SourceChangeEvent{{
            RepoID: repo,
            Kind:   live.ChangeBulkUpdate,
            Source: "coalescer",
        }}
    }
    sort.Slice(result, func(i, j int) bool { return result[i].Path < result[j].Path })
    return result
}

// MergeChange applies the SPEC §16.2 merge table. Returns (merged, keep)
// where keep=false means the path should drop out (created+deleted no-op).
func MergeChange(prev, cur live.SourceChangeEvent) (live.SourceChangeEvent, bool) {
    switch {
    case prev.Kind == live.ChangeFileModified && cur.Kind == live.ChangeFileModified:
        return cur, true
    case prev.Kind == live.ChangeFileCreated && cur.Kind == live.ChangeFileModified:
        prev.ObservedAt = cur.ObservedAt
        return prev, true
    case prev.Kind == live.ChangeFileCreated && cur.Kind == live.ChangeFileDeleted:
        return live.SourceChangeEvent{}, false
    case prev.Kind == live.ChangeFileModified && cur.Kind == live.ChangeFileDeleted:
        return cur, true
    case prev.Kind == live.ChangeFileDeleted && cur.Kind == live.ChangeFileCreated:
        return live.SourceChangeEvent{
            Kind:       live.ChangeFileModified,
            Path:       cur.Path,
            RepoID:     cur.RepoID,
            Source:     cur.Source,
            ObservedAt: cur.ObservedAt,
        }, true
    case prev.Kind == live.ChangeFileRenamed && cur.Kind == live.ChangeFileModified && cur.Path == prev.Path:
        // rename already covers the new-side path; keep rename, advance time
        prev.ObservedAt = cur.ObservedAt
        return prev, true
    default:
        return cur, true
    }
}

func semanticRepoID(events []live.SourceChangeEvent) (out interface{ String() string }) {
    // Defensive: pick first non-empty RepoID. Returns zero if none.
    for _, ev := range events {
        if string(ev.RepoID) != "" {
            return interface{ String() string }(nil) // FIXME — see note
        }
    }
    return nil
}

package live

import (
    "context"
    "errors"
    "io/fs"
    "os"

    "github.com/agenthands/helix/internal/semantic"
)

// FileHashLookup is the read-side store API the classifier consults to decide
// whether a path was previously known. The classifier MUST NOT import store
// directly to keep the dependency arrow one-way; the daemon wires this in
// from the live service constructor.
type FileHashLookup interface {
    EffectiveContentHash(ctx context.Context, repoID semantic.RepoID, path string) (string, bool, error)
}

// FileHasher computes the content hash for a path. xxhash64 in production
// (matches Phase 59 stable-ID hash, D-05 invariant).
type FileHasher func(absPath string) (string, error)

// ClassifyPathChange decides the SourceChangeKind for a single path. Single
// owner of kind decisions per D-01; sources only emit paths.
//
// Returns (kind, true) on a real change; (zero, false) on a no-op (e.g.,
// path missing AND unknown to the store).
func ClassifyPathChange(
    ctx context.Context,
    repoID semantic.RepoID,
    absPath string,
    lookup FileHashLookup,
    hasher FileHasher,
    source ChangeSource,
) (SourceChangeKind, bool, error) {
    if source == ChangeSourceHelixEdit {
        return ChangeHelixEdit, true, nil
    }
    _, statErr := os.Stat(absPath)
    pathExists := statErr == nil
    pathMissing := errors.Is(statErr, fs.ErrNotExist)
    if !pathExists && !pathMissing && statErr != nil {
        return "", false, statErr
    }

    _, known, err := lookup.EffectiveContentHash(ctx, repoID, absPath)
    if err != nil {
        return "", false, err
    }
    switch {
    case pathExists && !known:
        return ChangeFileCreated, true, nil
    case pathExists && known:
        return ChangeFileModified, true, nil
    case pathMissing && known:
        return ChangeFileDeleted, true, nil
    default:
        return "", false, nil // path missing, never known → no-op
    }
}
```

package handler

import (
    "context"
    "fmt"

    "github.com/agenthands/helix/internal/semantic"
    "github.com/agenthands/helix/internal/semantic/live"
    "github.com/agenthands/helix/internal/semantic/scheduler"
)

// OverlayWriter is the subset of store.Store the handler needs.
type OverlayWriter interface {
    BeginOverlayTx(ctx context.Context, repoID string) (OverlayTx, error)
}

// OverlayTx is the subset of store.OverlayTx the handler uses.
type OverlayTx interface {
    Epoch() uint64
    UpsertOverlayFile(ctx context.Context, path, contentHash string) error
    Commit() error
    Rollback() error
}

// IncrementalScheduler is the subset of scheduler.ExtractionScheduler the
// bulk handler invokes.
type IncrementalScheduler interface {
    ScheduleIncremental(workspaceID semantic.WorkspaceID, changes []scheduler.FileChange) scheduler.JobID
}

type Handler struct {
    Store   OverlayWriter
    Hasher  func(absPath string) (string, error)
    Sched   IncrementalScheduler
    Logger  Logger // slog or shim
}

type Logger interface {
    Warn(msg string, args ...any)
    Info(msg string, args ...any)
}

// Dispatch dispatches a single coalesced event to the appropriate handler.
// Per-event errors are logged but do not abort the batch (D-02 invariant).
// Empty batches must be filtered by the caller BEFORE Dispatch is called —
// see the Coalescer flush path.
func (h *Handler) Dispatch(ctx context.Context, ev live.SourceChangeEvent) error {
    switch ev.Kind {
    case live.ChangeFileCreated, live.ChangeFileModified, live.ChangeHelixEdit:
        return h.UpdateChangedFile(ctx, ev.RepoID, ev.Path)
    case live.ChangeFileDeleted:
        return h.HandleFileDeleted(ctx, ev.RepoID, ev.Path)
    case live.ChangeFileRenamed:
        return h.HandleFileRenamed(ctx, ev.RepoID, ev.OldPath, ev.Path)
    case live.ChangeBulkUpdate:
        return h.HandleBulkUpdate(ctx, ev.RepoID)
    default:
        return fmt.Errorf("handler.Dispatch: unknown kind %q", ev.Kind)
    }
}

func (h *Handler) UpdateChangedFile(ctx context.Context, repoID semantic.RepoID, path string) error {
    hash, err := h.Hasher(path)
    if err != nil {
        return fmt.Errorf("UpdateChangedFile: hash %s: %w", path, err)
    }
    tx, err := h.Store.BeginOverlayTx(ctx, string(repoID))
    if err != nil {
        return fmt.Errorf("UpdateChangedFile: begin tx: %w", err)
    }
    if err := tx.UpsertOverlayFile(ctx, path, hash); err != nil {
        _ = tx.Rollback()
        return fmt.Errorf("UpdateChangedFile: upsert: %w", err)
    }
    return tx.Commit()
}

    if err != nil {

package coalescer

import (
    "context"
    "sync"
    "time"

    "github.com/agenthands/helix/internal/semantic/live"
    "github.com/agenthands/helix/internal/workspace"
)

// EventHandler is the consumer side: receives one coalesced event per call.
type EventHandler interface {
    Dispatch(ctx context.Context, ev live.SourceChangeEvent) error
}

type Config struct {
    DebounceMs           time.Duration
    MaxBatchDelayMs      time.Duration
    BulkChangeThreshold  int
    QueueSize            int
}

type Logger interface {
    Warn(msg string, args ...any)
}

// Coalescer is per-workspace single-goroutine debounce + flush + dispatch.
type Coalescer struct {
    workspaceID workspace.WorkspaceKey
    cfg         Config
    handler     EventHandler
    logger      Logger
    in          chan live.SourceChangeEvent
    drops       atomic.Uint64

    mu          sync.Mutex
    pending     map[string]live.SourceChangeEvent
    timer       *time.Timer
    maxTimer    *time.Timer
}

func New(wsID workspace.WorkspaceKey, cfg Config, handler EventHandler, logger Logger) *Coalescer {
    if cfg.QueueSize <= 0 {
        cfg.QueueSize = 1024
    }
    if cfg.DebounceMs <= 0 {
        cfg.DebounceMs = 250 * time.Millisecond
    }
    if cfg.MaxBatchDelayMs <= 0 {
        cfg.MaxBatchDelayMs = 1500 * time.Millisecond
    }
    if cfg.BulkChangeThreshold <= 0 {
        cfg.BulkChangeThreshold = 200
    }
    return &Coalescer{
        workspaceID: wsID,
        cfg:         cfg,
        handler:     handler,
        logger:      logger,
        in:          make(chan live.SourceChangeEvent, cfg.QueueSize),
        pending:     make(map[string]live.SourceChangeEvent),
    }
}

// Enqueue is non-blocking. Drops with a counter increment when the input
// channel is full (D-02 invariant: producers MUST NOT block).
func (c *Coalescer) Enqueue(ev live.SourceChangeEvent) {
    select {
    case c.in <- ev:
    default:
        c.drops.Add(1)
        c.logger.Warn("coalescer queue full; dropped event",
            "workspace", c.workspaceID, "path", ev.Path)
    }
}

// Run drains the input channel until ctx is cancelled. Single goroutine per
// workspace.
func (c *Coalescer) Run(ctx context.Context) error {
    flush := c.makeFlush(ctx)
    for {
        select {
        case <-ctx.Done():
            c.mu.Lock()
            if c.timer != nil { c.timer.Stop() }
            if c.maxTimer != nil { c.maxTimer.Stop() }
            c.mu.Unlock()
            return ctx.Err()
        case ev := <-c.in:
            c.accept(ev, flush)
        }
    }
}

func (c *Coalescer) accept(ev live.SourceChangeEvent, flush func()) {
    c.mu.Lock()
    defer c.mu.Unlock()
    key := ev.Path
    if ev.Kind == live.ChangeFileRenamed { key = ev.OldPath + "→" + ev.Path }
    c.pending[key] = ev // last-write-wins at accept-time; CoalesceEvents re-applies merge rules at flush
    if c.timer != nil { c.timer.Stop() }
    c.timer = time.AfterFunc(c.cfg.DebounceMs, flush)
    if c.maxTimer == nil {
        c.maxTimer = time.AfterFunc(c.cfg.MaxBatchDelayMs, flush)
    }
}

func (c *Coalescer) makeFlush(ctx context.Context) func() {
    return func() {
        c.mu.Lock()
        if len(c.pending) == 0 {
            c.mu.Unlock()
            return
        }
        snapshot := make([]live.SourceChangeEvent, 0, len(c.pending))
        for _, v := range c.pending {
            snapshot = append(snapshot, v)
        }
        c.pending = make(map[string]live.SourceChangeEvent)
        if c.maxTimer != nil { c.maxTimer.Stop(); c.maxTimer = nil }
        c.mu.Unlock()

        merged := CoalesceEvents(snapshot, c.cfg.BulkChangeThreshold)
        if len(merged) == 0 {
            return // D-04 invariant: no-op flush does not call dispatch
        }
        for _, ev := range merged {
            if err := c.handler.Dispatch(ctx, ev); err != nil {
                c.logger.Warn("coalescer: dispatch error",
                    "workspace", c.workspaceID, "path", ev.Path, "err", err)
                // continue — D-02 invariant: per-event errors do not abort batch
            }
        }
    }
}

type Service struct {

    handler     coalescer.EventHandler

    repoIDFor   func(workspace.WorkspaceKey) semantic.RepoID

    mu          sync.RWMutex
    coalescers  map[workspace.WorkspaceKey]*coalescer.Coalescer

}

func NewService(cfg coalescer.Config, handler coalescer.EventHandler, classifier func(ctx context.Context, repoID semantic.RepoID, path string, src ChangeSource) (SourceChangeKind, bool, error), repoIDFor func(workspace.WorkspaceKey) semantic.RepoID, logger coalescer.Logger) *Service {
    return &Service{
        cfg:         cfg,
        handler:     handler,
        classifier:  classifier,
        repoIDFor:   repoIDFor,
        logger:      logger,
        now:         time.Now,
        coalescers:  make(map[workspace.WorkspaceKey]*coalescer.Coalescer),
        runCancels:  make(map[workspace.WorkspaceKey]context.CancelFunc),
    }
}

// Start spins up a coalescer goroutine for ws. Idempotent.
func (s *Service) Start(ctx context.Context, ws workspace.WorkspaceKey) {
    s.mu.Lock()
    defer s.mu.Unlock()
    if _, ok := s.coalescers[ws]; ok { return }
    c := coalescer.New(ws, s.cfg, s.handler, s.logger)
    cctx, cancel := context.WithCancel(ctx)
    s.coalescers[ws] = c
    s.runCancels[ws] = cancel
    go func() { _ = c.Run(cctx) }()
}

// Stop cancels a per-workspace coalescer.
func (s *Service) Stop(ws workspace.WorkspaceKey) {
    s.mu.Lock()
    cancel := s.runCancels[ws]
    delete(s.coalescers, ws)
    delete(s.runCancels, ws)
    s.mu.Unlock()
    if cancel != nil { cancel() }
}

// OnEdit implements kernel.EditNotifier. Fire-and-forget: classifies
// (synchronously, but ChangeSourceHelixEdit short-circuits without I/O) and
// non-blocking enqueues into the per-workspace coalescer.
func (s *Service) OnEdit(ctx context.Context, ws workspace.WorkspaceKey, paths []string) error {
    return s.signal(ctx, WorkspaceChangeSignal{
        WorkspaceID: ws,
        Paths:       paths,
        Source:      ChangeSourceHelixEdit,
        ObservedAt:  s.now(),
    })
}

// OnWorkspaceChanged is the public single entry the watcher and scanner call.
func (s *Service) OnWorkspaceChanged(ctx context.Context, sig WorkspaceChangeSignal) error {
    return s.signal(ctx, sig)
}

func (s *Service) signal(ctx context.Context, sig WorkspaceChangeSignal) error {
    s.mu.RLock()
    c := s.coalescers[sig.WorkspaceID]
    s.mu.RUnlock()
    if c == nil {
        // Workspace not yet started; drop the signal. Log at debug.
        return nil
    }
    repoID := s.repoIDFor(sig.WorkspaceID)
    for _, path := range sig.Paths {
        kind, ok, err := s.classifier(ctx, repoID, path, sig.Source)
        if err != nil {
            s.logger.Warn("classifier error", "path", path, "err", err)
            continue
        }
        if !ok {
            continue
        }
        c.Enqueue(SourceChangeEvent{
            RepoID:     repoID,
            Kind:       kind,
            Path:       path,
            Source:     sig.Source,
            ObservedAt: sig.ObservedAt,
        })
    }
    return nil
}

// Compile-time assertion: Service satisfies kernel.EditNotifier.
var _ kernel.EditNotifier = (*Service)(nil)
