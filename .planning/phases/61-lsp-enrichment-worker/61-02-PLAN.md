       ```go
       func (c *Cascade) Run(ctx, job, lang, budget, foregroundBusy) Outcome {
           tx, err := c.Store.BeginOverlayTx(ctx, string(job.RepoID))
           if err != nil { /* log; return OutcomeDropped */ }
           defer func() {
               if !committed { _ = tx.Rollback() }
           }()

           checkBoundary := func(stepName string) Outcome {
               if foregroundBusy() {
                   _ = tx.MarkFileSemanticPending(ctx, job.Path, "preempted")
                   return OutcomePartialPreempted
               }
               if !budget.HasRemainingTime(time.Now()) {
                   _ = tx.MarkFileSemanticPending(ctx, job.Path, "budget exhausted")
                   return OutcomePartialBudget
               }
               return ""
           }

           // Step 1: documentSymbol
           if o := checkBoundary("documentSymbol"); o != "" { commit(tx); return o }
           syms, err := callDocumentSymbol(ctx, c.Lease, job.Path)
           if isMethodNotFound(err) { c.Capabilities.MarkUnsupported(lang, "documentSymbol"); slog.Debug("..."); /* fall through */ }
           else if err != nil {
               // W1 — cascade does NOT release lease; that responsibility is the
               // Manager's per B2 fix-path-A. Manager observes the OutcomePartial-
               // LSPUnavail outcome (via metrics) and discards the cached handle
               // on its own circuit-breaker policy.
               _ = tx.MarkFileSemanticPending(ctx, job.Path, "lsp_unavailable")
               commit(tx)
               return OutcomePartialLSPUnavail
           }
           if syms != nil { _ = tx.UpsertSymbols(ctx, job.Path, mapSymbols(syms)) }

           // Step 2: drain published diagnostics
           if o := checkBoundary("diagnostics"); o != "" { commit(tx); return o }
           diags := drainDiagnostics(c.Lease, job.Path)
           _ = tx.UpsertDiagnostics(ctx, job.Path, diags)

           // Step 3: prioritize + per-symbol inner loop
           prioritized := prioritizeSymbols(syms, budget)
           for _, sym := range prioritized {
               // W10 — per-symbol budget consumption granularity:
               // ConsumeSymbol once per symbol; ConsumeReferences(len(refs))
               // once per symbol BEFORE the per-reference loop. Both must
               // return ok=true or cascade aborts to OutcomePartialBudget.
               if !budget.ConsumeSymbol() {
                   _ = tx.MarkFileSemanticPending(ctx, job.Path, "budget exhausted"); commit(tx); return OutcomePartialBudget
               }

               if o := checkBoundary("hover"); o != "" { commit(tx); return o }
               doHover(ctx, c.Lease, sym, tx)  // edges land with Confidence=1.0, ValidationState="validated", Source="lsp.hover" (W6 — typed assertion in test)

               if !c.Capabilities.IsUnsupported(lang, "callHierarchy") {
                   if o := checkBoundary("callHierarchy"); o != "" { commit(tx); return o }
                   doCallHierarchy(ctx, c.Lease, sym, budget.CallDepth(), tx)
               }
       ```
