       ```go
       package lspenrich

       import (
           "sync"
           "sync/atomic"
       )

       // Status is a read-only snapshot of the enrichment Manager's state.
       // Phase 61 D-09: ships the accessor; Phase 65 wires it into get_health.
       //
       // W11 — LastErrorPerLanguage is map[string]string for v1: the value
       // is err.Error() of the most recent error class (e.g. "circuit_open",
       // "timeout"). Phase 65 may evolve to a structured error type without
       // breaking this surface.
       type Status struct {
           LaneDepths           map[Lane]int
           InFlight             int
           FilesEnriched        uint64
           FilesDropped         uint64
           FilesPreempted       uint64
           FilesPending         uint64
           LastErrorPerLanguage map[string]string
       }

       type statusTracker struct {
           filesEnriched    atomic.Uint64
           filesDropped     atomic.Uint64
           filesPreempted   atomic.Uint64
           filesPending     atomic.Uint64
           inFlight         atomic.Int64
           mu               sync.RWMutex
           lastErrorPerLang map[string]string
       }

       func newStatusTracker() *statusTracker {
           return &statusTracker{lastErrorPerLang: make(map[string]string)}
       }

       func (s *statusTracker) recordOutcome(outcome Outcome) {
           switch outcome {
           case OutcomeApplied:
               s.filesEnriched.Add(1)
           case OutcomeDropped:
               s.filesDropped.Add(1)
           case OutcomePartialPreempted:
               s.filesPreempted.Add(1)
               s.filesPending.Add(1)
           case OutcomePartialBudget:
               // W7 — partial_budget is "applied with caveat": the file IS
               // queryable (partial facts committed) BUT also pending re-
               // enrichment. Surface both counters so get_health consumers
               // can compute fully-enriched = enriched - pending.
               s.filesEnriched.Add(1)
               s.filesPending.Add(1)
           case OutcomePartialLSPUnavail:
               s.filesPending.Add(1)
           }
       }

       func (s *statusTracker) recordError(lang, msg string) {
           s.mu.Lock()
           defer s.mu.Unlock()
           s.lastErrorPerLang[lang] = msg
       }

       func (s *statusTracker) snapshot(laneDepths map[Lane]int) Status {
           s.mu.RLock()
           lpe := make(map[string]string, len(s.lastErrorPerLang))
           for k, v := range s.lastErrorPerLang { lpe[k] = v }
           s.mu.RUnlock()
           return Status{
               LaneDepths:           laneDepths,
               InFlight:             int(s.inFlight.Load()),
               FilesEnriched:        s.filesEnriched.Load(),
               FilesDropped:         s.filesDropped.Load(),
               FilesPreempted:       s.filesPreempted.Load(),
               FilesPending:         s.filesPending.Load(),
               LastErrorPerLanguage: lpe,
           }
       }
       ```

       // Manager owns the enrichment workers AND the per-(wsKey, lang) lease

       type Manager struct {
           queue     *LaneQueue

           readiness ReadinessProbe

           metrics   MetricsSink

           // Released on OnWorkspaceDeactivate (workspace) or Stop (all).

           // so the next job retries.
           leasesMu sync.Mutex
           leases   map[wsLangKey]*lspool.WorkerLease
       }

       func NewManager(
           queue *LaneQueue,
           acquirer LeaseAcquirer,
           store OverlayStore,
           readiness ReadinessProbe,
           cfg semantic.LSPEnrichmentConfig,
           metrics MetricsSink,
           logger *slog.Logger,

           if metrics == nil { metrics = noopMetrics{} }
           if logger == nil { logger = slog.Default() }
           return &Manager{
               queue: queue, acquirer: acquirer, store: store,
               readiness: readiness, cfg: cfg, metrics: metrics, logger: logger,
               tracker: newStatusTracker(),
               leases: make(map[wsLangKey]*lspool.WorkerLease),
           }
       }

       // **B2 — Manager satisfies P02's Worker.LeaseProvider interface.**
       // AcquireFor returns a cached *WorkerLease for (wsKey, lang) or
       // lazily acquires one (sessionID = "lsp-enrichment:<repo>:<lang>";
       // dirty=false; clean lease for share-until-dirty reuse with
       // foreground sessions). ErrCircuitOpen surfaces the error WITHOUT
       // caching the failed handle.
       //
       // Concurrency: a sync.Mutex around the map + acquire is sufficient
       // here because lease acquisition is rare relative to job throughput
       // (one Acquire per (workspace, language) per process lifetime under
       // normal operation). For 100 concurrent AcquireFor calls on a
       // cold key, the lock funnels to a single AcquireLease call.
       func (m *Manager) AcquireFor(ctx context.Context, ws workspace.WorkspaceKey, lang string) (*lspool.WorkerLease, error) {
           key := wsLangKey{ws: ws, lang: lang}

           m.leasesMu.Lock()
           if cached, ok := m.leases[key]; ok && cached != nil {
               m.leasesMu.Unlock()
               return cached, nil
           }
           m.leasesMu.Unlock()

           // Acquire OUTSIDE the map lock to avoid blocking other (wsKey, lang)
           // pairs. We re-check after acquire to handle the race.
           sessionID := fmt.Sprintf("lsp-enrichment:%s:%s", repoIDFromWS(ws), lang)
           lease, err := m.acquirer.AcquireLease(ctx, sessionID, ws, false /* clean */)
           if err != nil {
               // B2 — ErrCircuitOpen and other errors are NOT cached. Slot
               // remains empty; next job retries.
               return nil, err
           }

           m.leasesMu.Lock()
           defer m.leasesMu.Unlock()
           if existing, ok := m.leases[key]; ok && existing != nil {
               // Race: another goroutine cached one between our unlock and
               // re-lock. Release ours, return theirs.
               lease.Release()
               return existing, nil
           }
           m.leases[key] = lease
           return lease, nil
       }

       // **B2 — OnWorkspaceDeactivate releases all cached leases for ws.**
       // Wired in P03-T5 from the daemon's existing workspace-deactivate path.
       func (m *Manager) OnWorkspaceDeactivate(ws workspace.WorkspaceKey) {
           m.leasesMu.Lock()
           defer m.leasesMu.Unlock()
           for k, lease := range m.leases {
               if k.ws == ws {
                   if lease != nil {
                       lease.Release()
                   }
                   delete(m.leases, k)
               }
           }
       }

       // Run starts MaxConcurrentWorkers worker goroutines.
       func (m *Manager) Run(ctx context.Context) error {
           ctx, m.cancel = context.WithCancel(ctx)
           wrappedMetrics := &trackedMetrics{inner: m.metrics, tracker: m.tracker}
           w := &Worker{
               Queue:     m.queue,
               Leases:    m,             // B2 — Manager IS the LeaseProvider.
               Acquirer:  m.acquirer,    // For ForegroundBusy only.
               Store:     m.store,
               Readiness: m.readiness,
               BudgetCfg: m.cfg,
               Metrics:   wrappedMetrics,
               Logger:    m.logger,
           }
           m.logger.Info("lsp-enrichment manager starting",
               "max_concurrent_workers", m.cfg.MaxConcurrentWorkers,
               "yield_check_window_ms", m.cfg.YieldCheckWindowMs,
               "timeout_per_file", m.cfg.TimeoutPerFile,
           )
           g, gctx := errgroup.WithContext(ctx)
           g.Go(func() error { return w.RunN(gctx, m.cfg.MaxConcurrentWorkers) })
           err := g.Wait()
           // **B2** — on Run exit, release all cached leases.
           m.releaseAll()
           return err
       }

       func (m *Manager) Stop() {
           if m.cancel != nil { m.cancel() }
           m.releaseAll()
       }

       func (m *Manager) releaseAll() {
           m.leasesMu.Lock()
           defer m.leasesMu.Unlock()
           for k, lease := range m.leases {
               if lease != nil { lease.Release() }
               delete(m.leases, k)
           }
       }

       func (m *Manager) Status() Status {
           depths := map[Lane]int{
               LaneHigh:       m.queue.Depth(LaneHigh),
               LaneBackground: m.queue.Depth(LaneBackground),
