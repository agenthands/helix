           cfg := semantic.LSPEnrichmentConfig{
               Enabled:              true,
               MaxConcurrentWorkers: 1,
               YieldCheckWindowMs:   200,
               TimeoutPerFile:       "10s",
               TimeoutTotal:         "60s",
               MaxSymbolsPerFile:    50,
               MaxReferencesPerSymbol: 100,
               MaxReferencesPerFile: 500,
               MaxCallHierarchyDepth: 2,
               MaxTypeHierarchyDepth: 2,
           }
           pool.SetYieldCheckWindow(200 * time.Millisecond)

           mgr := lspenrich.NewManager(
               laneQueue,
               lspenrich.NewPoolAcquirer(pool),
               store,
               lspenrich.NewPoolReadinessProbe(pool),
               cfg,
               recordingMetrics,
               slog.Default(),
           )

           ctx, cancel := context.WithTimeout(context.Background(), 30*time.Second)
           defer cancel()

           runErr := make(chan error, 1)
           go func() { runErr <- mgr.Run(ctx) }()
           defer mgr.Stop()

           // Step 1: enqueue ONE job.
           _ = laneQueue.EnqueueLane(lspenrich.LaneHigh,
               lspqueue.RevalidateFileJob{RepoID: "stress", Path: "main.go"})

           // Step 2-3: wait for the SLOW LS to receive call #2.

           wsKey := mockLS.WorkspaceKey()
           foregroundLease, err := pool.AcquireLease(ctx, "foreground-tool:X", wsKey, false)
           if err != nil { t.Fatalf("foreground AcquireLease: %v", err) }
           defer foregroundLease.Release()

           // Step 5: wait for cascade outcome (recordingMetrics signals on
           // every LSPEnrichmentTotal call).
           select {
           case outcome := <-recordingMetrics.OutcomeCh():
               if outcome.outcome != string(lspenrich.OutcomePartialPreempted) {
                   t.Fatalf("expected outcome=partial_preempted (B1 / acceptance #6); got %q", outcome.outcome)
               }
           case <-time.After(10 * time.Second):
               t.Fatalf("cascade did not return within timeout")
           }

           // Step 6 assertions:
           // a. partial_reason="preempted" stamped on the file.
           pr := store.GetPartialReason("stress", "main.go")
           if pr != "preempted" {
               t.Fatalf("expected partial_reason=preempted; got %q", pr)
           }
           // b. Only ~2 cascade steps' facts committed (count Upsert calls).
           commits := store.UpsertCallsFor("main.go")
           if commits < 1 || commits > 3 {
               t.Fatalf("expected 1-3 upsert batches (steps 1-2 committed); got %d", commits)
           }
       }
