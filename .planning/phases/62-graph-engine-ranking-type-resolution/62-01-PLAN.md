```go
package graph

type Options struct {
    Damping     float64         // default 0.85
    Epsilon     float64         // default 1e-6
    MaxIter     int             // default 100
    Personalize map[any]float64 // optional; nil → uniform teleport. Keys MUST be assertable to T at call site.
}

// internal/graph/pagerank.go
package graph

import (
    "cmp"
    "math"
    "sort"
)

// PageRank computes weighted PageRank deterministically. Inputs:
//   nodes  — full node-id slice (caller may pass unsorted; engine sorts a copy).
//   edges  — adjacency map: edges[src][dst] = weight.
//   opts   — convergence options. Damping<=0 → 0.85; Epsilon<=0 → 1e-6; MaxIter<=0 → 100.
// Determinism contract (D-03): every iteration uses a pre-sorted node slice,
// so map iteration order never affects output. Inner edge sums are
// commutative addition (order-independent).
func PageRank[T cmp.Ordered](nodes []T, edges map[T]map[T]float64, opts Options) map[T]float64

// RankNodes returns nodes sorted by score descending, with ties broken by
// sorted-key (= stable-key NodeID) order per GRAPH-03.
func RankNodes[T cmp.Ordered](nodes []T, edges map[T]map[T]float64, opts Options) []Ranked[T]

type Ranked[T cmp.Ordered] struct {
    Node  T
    Score float64
}
```

`internal/repomap/pagerank.go` keeps its existing `(g *FileGraph).PageRank(damping, epsilon, maxIter, personalization)` and `(g *FileGraph).RankFiles(...)` public methods. Bodies delegate to `graph.PageRank` / `graph.RankNodes` after assembling the inputs under `g.mu.RLock()`.
