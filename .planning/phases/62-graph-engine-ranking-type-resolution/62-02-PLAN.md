// internal/semantic/graph/repair.go
package graph

type NodeID = uint64 // alias for the existing semantic NodeID

type GraphEdge struct {
    SrcNodeID, DstNodeID NodeID
    EdgeKind             string  // e.g., "CALLS", "RESOLVES_TO", "USES_TYPE"
    Confidence           float64
    Source               string  // "lsp.<call>", "comment.<kind>", ...
    ValidationState      string  // "validated", "unresolved", ...
    Weight               float64
}

type GraphRepair struct {
    RemovedNodes        []NodeID
    DirtyNodes          []NodeID
    UpsertedEdges       []GraphEdge
    RemovedEdges        []GraphEdge
    InvalidatedIncoming []NodeID
    InvalidatedOutgoing []NodeID
}

func (r GraphRepair) IsEmpty() bool

// FileFactDiff is the post-commit diff the handler hands to ComputeGraphRepair.
type FileFactDiff struct {
    RemovedSymbols []SymbolDiff
    ChangedSymbols []SymbolDiff   // includes flags: SignatureChanged, ExportedChanged, KindChanged, StableKeyChanged, BodyOnlyChanged
    AddedSymbols   []SymbolDiff
    AddedEdges     []GraphEdge
    RemovedEdges   []GraphEdge
}

type SymbolDiff struct {
    NodeID            NodeID
    SignatureChanged  bool
    ExportedChanged   bool
    KindChanged       bool
    StableKeyChanged  bool
    BodyOnlyChanged   bool
}

func ComputeGraphRepair(diff FileFactDiff) GraphRepair

type ScoreStatus string

const (
    ScoreStatusExact       ScoreStatus = "exact"
    ScoreStatusApproximate ScoreStatus = "approximate"
    ScoreStatusStale       ScoreStatus = "stale"
    ScoreStatusMissing     ScoreStatus = "missing"
)

// computeScoreStatus is the closed-enum decision at read time (D-07).
func computeScoreStatus(currentGV, rowGV uint64, hasRow bool, approximate bool) ScoreStatus

// internal/semantic/graph/apply_repair.go
type Engine struct {
    store         RepairStore       // narrow tx seam (interface; production = *store.Store)
    logger        *slog.Logger
    metrics       MetricsSink
    notifyVersion chan<- GraphVersionAdvance // optional channel for RankScheduler (P03 wires; see GraphVersionAdvance below)
}

// GraphVersionAdvance is the typed payload published by ApplyRepair to the
// RankScheduler. ChangedNodes lets P03 derive the incremental "changed nodes"
// directly from the channel — no fallback to "score rows currently marked
// stale" (W3).
type GraphVersionAdvance struct {
    RepoID       string
    Version      uint64
    ChangedNodes []NodeID
}

// ApplyRepair is the SINGLE site that bumps graph_version (D-06).
// Caller (handler) acquires the per-workspace mutex via store.overlayLockFor(repoID)
// before calling — see apply_repair.go for the lock-ordering contract.
func (e *Engine) ApplyRepair(ctx context.Context, repoID string, repair GraphRepair) (newGV uint64, bumped bool, err error)

// internal/semantic/graph/ranker.go
type Ranker interface {
    Rank(ctx context.Context, req RankRequest) (RankResponse, error)
    Status(ctx context.Context, repoID string) (RankStatus, error)
}

type RankRequest struct {
    RepoID         string
    Projection     string             // "call_graph"
    SeedFiles      []string           // GRAPH-02 personalized PR seed
    SeedSymbolIDs  []NodeID           // GRAPH-02
    Limit          int
}

type RankResponse struct {
    GraphVersion    uint64
    EnrichmentLevel string                  // from cascade: "skeleton", "validated", ...
    Nodes           []RankedNode
}

type RankedNode struct {
    NodeID      NodeID
    Score       float64
    ScoreStatus ScoreStatus
}

type RankStatus struct {
    GraphVersion uint64
    PerProjection map[string]ProjectionStatus
}

type ProjectionStatus struct {
    NodeCount     int
    ExactCount    int
    StaleCount    int
    ApproxCount   int
    MissingCount  int
}

// UpsertEdgesWithMerge enforces the (src_node_id, dst_node_id, edge_kind) merge

}

// BumpGraphVersion advances semantic_live_overlay_meta.graph_version under
// the existing per-tx mutex. Returns the new value. ONLY ApplyRepair calls this.

// CurrentGraphVersion reads the current value (no advance). For score_status
// computation at read time.
