       // sortedNodeIDs returns the keys of a NodeID-keyed map in ascending order.

       ```go
       func ComputeFrontier(changed []NodeID,
           adjacencyOut, adjacencyIn map[NodeID]map[NodeID]float64,
           maxNodes int) (frontier []NodeID, overflow bool) {

           seen := make(map[NodeID]struct{}, len(changed)*4)
           // Sort `changed` first — caller might pass it unsorted.
           changedSorted := append([]NodeID(nil), changed...)
           sort.Slice(changedSorted, func(i, j int) bool { return changedSorted[i] < changedSorted[j] })

           for _, n := range changedSorted {
               seen[n] = struct{}{}
               if len(seen) > maxNodes { return nil, true }
               // Outgoing 1-hop — iterate adjacency with sortedNodeIDs to keep build deterministic.
               for _, dst := range sortedNodeIDs(adjacencyOut[n]) {
                   seen[dst] = struct{}{}
                   if len(seen) > maxNodes { return nil, true }
               }
               // Incoming 1-hop.
               for _, src := range sortedNodeIDs(adjacencyIn[n]) {
                   seen[src] = struct{}{}
                   if len(seen) > maxNodes { return nil, true }
               }
           }
           return sortedNodeIDs(seen), false
       }
       ```
