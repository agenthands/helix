    ```go
    package store

    import (
      "context"
      "database/sql"
      "fmt"

      "github.com/agenthands/helix/internal/semantic/graph"
    )

    // QueryEffectiveAdjacency: SELECT-UNION-ALL across snapshot + overlay (non-tombstoned)
    // for (repo, projection). Reads under no overlay/workspace lock per scheduler_store.go:48-56.
    func (s *Store) QueryEffectiveAdjacency(ctx context.Context, repoID, projection string) (
        out, in map[graph.NodeID]map[graph.NodeID]float64, err error,
    ) {
      out = make(map[graph.NodeID]map[graph.NodeID]float64)
      in  = make(map[graph.NodeID]map[graph.NodeID]float64)
      const q = `
        SELECT src, dst, weight FROM semantic_edges
          WHERE repo_id = ? AND projection = ?
        UNION ALL
        SELECT src, dst, weight FROM semantic_live_overlay_edges
          WHERE repo_id = ? AND projection = ? AND tombstone = false
      `
      rows, err := s.db.QueryContext(ctx, q, repoID, projection, repoID, projection)
      if err != nil { return nil, nil, fmt.Errorf("effective adjacency query: %w", err) }
      defer rows.Close()
      for rows.Next() {
        var src, dst string; var w float64
        if err := rows.Scan(&src, &dst, &w); err != nil { return nil, nil, err }
        if out[graph.NodeID(src)] == nil { out[graph.NodeID(src)] = make(map[graph.NodeID]float64) }
        if in[graph.NodeID(dst)]  == nil { in[graph.NodeID(dst)]  = make(map[graph.NodeID]float64) }
        out[graph.NodeID(src)][graph.NodeID(dst)] = w
        in[graph.NodeID(dst)][graph.NodeID(src)] = w
      }
      return out, in, rows.Err()
    }

    // CountStaleScoreRows: lock-free scalar count.
    func (s *Store) CountStaleScoreRows(ctx context.Context, repoID, projection string) (stale, total int, err error) {
      const q = `
        SELECT
          COALESCE(SUM(CASE WHEN status='stale' THEN 1 ELSE 0 END), 0) AS stale,
          COUNT(*) AS total
        FROM semantic_pagerank_scores
        WHERE repo_id = ? AND projection = ?
      `
      err = s.db.QueryRowContext(ctx, q, repoID, projection).Scan(&stale, &total)
      if err == sql.ErrNoRows { return 0, 0, nil }
      return stale, total, err
    }

    // MarkAllScoreRowsStale: idempotent UPDATE.
    func (s *Store) MarkAllScoreRowsStale(ctx context.Context, repoID, projection string) error {
      const q = `UPDATE semantic_pagerank_scores SET status='stale' WHERE repo_id=? AND projection=?`
      _, err := s.db.ExecContext(ctx, q, repoID, projection)
      return err
    }

    // LatestCommittedSnapshot: highest committed snapshot id; 0 + nil when none.
    func (s *Store) LatestCommittedSnapshot(ctx context.Context, repoID string) (uint64, error) {
      const q = `SELECT COALESCE(MAX(snapshot_id), 0) FROM semantic_snapshots WHERE repo_id=? AND status='committed'`
      var id uint64
      err := s.db.QueryRowContext(ctx, q, repoID).Scan(&id)
      if err == sql.ErrNoRows { return 0, nil }
      return id, err
    }
    ```
