    Pseudocode:

    ```
    var ErrModeMustBeResolved = errors.New("IndexRunner.Run requires resolved mode (full|incremental); call ResolveAuto first")

    func (r *IndexRunner) Run(ctx, ws, mode, maxMs) (IndexResult, error) {
      if mode == "auto" || mode == "" {
        return IndexResult{}, ErrModeMustBeResolved
      }
      key := ws.RepoRoot() + "|" + mode  // singleflight key uses the RESOLVED mode

      deadline := time.Duration(maxMs) * time.Millisecond
      if deadline <= 0 || deadline > r.timeout { deadline = r.timeout }

      callCtx, cancel := context.WithTimeout(ctx, deadline)
      defer cancel()
      startedAt := time.Now()

      ch := r.sf.DoChan(key, func() (any, error) {
        bgCtx, bgCancel := context.WithCancel(context.Background())  // NOT request ctx (D-04)
        st := &buildState{startedAt: time.Now(), mode: mode, cancel: bgCancel, done: make(chan struct{})}
        r.inFlight.Store(ws.RepoRoot(), st)
        defer r.inFlight.Delete(ws.RepoRoot())
        defer close(st.done)
        return r.buildFn(bgCtx, ws, mode, st)
      })

      select {
      case v := <-ch:
        var res IndexResult
        if v.Val != nil { res = v.Val.(IndexResult) }
        res.DurationMs = time.Since(startedAt).Milliseconds()
        if v.Err != nil {
          res.Status = IndexStatusFailed
          return res, v.Err
        }
        res.Status = IndexStatusCommitted
        return res, nil
      case <-callCtx.Done():
        // Read in-flight progress for the partial result.
        var partial IndexResult
        if v, ok := r.inFlight.Load(ws.RepoRoot()); ok {
          st := v.(*buildState)
          partial.SnapshotID = st.snapshotID
          partial.FilesIndexed = st.filesIndexed.Load()
          partial.FilesReused = st.filesReused.Load()
        }
        partial.Partial = true
        partial.Status = IndexStatusBuilding
        partial.Freshness = FreshnessStale
        partial.DurationMs = deadline.Milliseconds()
        return partial, nil
      }
    }

    func (r *IndexRunner) ResolveAuto(ctx, ws) string {
      latest, err := r.store.LatestCommittedSnapshot(ctx, ws.RepoID())
      if err != nil || latest == 0 { return "full" }
      return "incremental"
    }
    ```
