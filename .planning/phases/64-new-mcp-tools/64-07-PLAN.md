       package retrieval
       import "sort"
       type FusedCandidate struct {
         SymbolID    string
         Score       float64
         GraphVersion uint64
         TextRank    int
         GraphRank   int
         MatchedTerms []string
       }
       // Fuse merges text + graph rankings via weighted RRF. Phase 62 sort-before-iterate.
       func Fuse(text []TextRank, graph []GraphRank, cfg RRFConfig, gvLookup func(symbolID string) uint64) []FusedCandidate {
         scores := make(map[string]float64)
         textRankByID := make(map[string]int)
         graphRankByID := make(map[string]int)
         for rank, t := range text {
           textRankByID[t.SymbolID] = rank + 1
           scores[t.SymbolID] += cfg.WText / float64(cfg.K + rank + 1)
         }
         for rank, g := range graph {
           graphRankByID[g.SymbolID] = rank + 1
           scores[g.SymbolID] += cfg.WGraph / float64(cfg.K + rank + 1)
         }
         // Materialize keys deterministically (sort first to avoid map-iteration nondeterminism BEFORE the final sort).
         ids := make([]string, 0, len(scores))
         for id := range scores { ids = append(ids, id) }
         sort.Strings(ids)
         out := make([]FusedCandidate, 0, len(ids))
         for _, id := range ids {
           out = append(out, FusedCandidate{
             SymbolID: id, Score: scores[id], GraphVersion: gvLookup(id),
             TextRank: textRankByID[id], GraphRank: graphRankByID[id],
           })
         }
         sort.SliceStable(out, func(i, j int) bool {
           if out[i].Score != out[j].Score { return out[i].Score > out[j].Score }
           if out[i].GraphVersion != out[j].GraphVersion { return out[i].GraphVersion > out[j].GraphVersion }
           return out[i].SymbolID < out[j].SymbolID
         })
         return out
       }

       func greedyPack(candidates []FusedCandidate, budget int, tokensPerCandidate func(FusedCandidate) int) []FusedCandidate {
         out := make([]FusedCandidate, 0, len(candidates))
         used := 0
         for _, c := range candidates {
           cost := tokensPerCandidate(c)
           if used + cost > budget { continue }  // skip; do not break — allows packing smaller followers
           out = append(out, c)
           used += cost
         }
         return out
       }
