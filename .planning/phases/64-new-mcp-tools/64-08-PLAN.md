       type semanticBundle struct {
         cfg       semanticConfig
         store     *semanticstore.Store
         scheduler *graph.RankScheduler
         queue     *lspenrich.LaneQueue
         live      *liveservice.Service
         logger    *slog.Logger
         metrics   *obs.Metrics

         mu        sync.Mutex
         engines   map[string]*retrieval.Engine     // key=ws.RepoRoot()

         skill     *semantic.SemanticSkill
       }

       func newSemanticBundle(cfg semanticConfig, store, scheduler, queue, live, logger, metrics) *semanticBundle {
         if store == nil { return nil }   // semantic disabled when store missing
         b := &semanticBundle{cfg: cfg, store: store, scheduler: scheduler, queue: queue, live: live, logger: logger, metrics: metrics, engines: make(map[string]*retrieval.Engine), recoveres: make(map[string]*retrieval.Recoverer)}
         buildFn := b.makeProductionBuildFn()
         b.runner = semantic.NewIndexRunner(b.storeAccessor(), buildFn, cfg.IndexTimeout)
         b.skill = semantic.GetSemanticSkill()
         if b.skill != nil {
           b.skill.SetStore(b.storeAccessor())
           b.skill.SetScheduler(b.schedulerAccessor())
           b.skill.SetQueue(b.queueAccessor())
           b.skill.SetLive(b.liveAccessor())
           b.skill.SetRunner(b.runner)
           b.skill.SetRetrieval(b.retrievalAccessor())
         }
         return b
       }

       func (b *semanticBundle) Run(ctx context.Context) error {
         <-ctx.Done()
         b.shutdown()
         return ctx.Err()
       }

       func (b *semanticBundle) shutdown() {
         b.mu.Lock(); defer b.mu.Unlock()
         for _, e := range b.engines { _ = e.Close() }
         if b.runner != nil { b.runner.Shutdown(context.Background()) }
       }

       func (b *semanticBundle) ensureRetrieval(ctx context.Context, ws workspace.WorkspaceKey) {
         b.mu.Lock(); defer b.mu.Unlock()
         key := ws.RepoRoot()
         if _, ok := b.engines[key]; ok { return }
         path := filepath.Join(ws.RepoRoot(), b.cfg.BleveSubdir)
         engine, err := retrieval.Open(path)
         if err != nil {
           engine, err = retrieval.New(path)
           if err != nil { b.logger.Warn("bleve init failed", "error", err); return }
         }
         b.engines[key] = engine
         rec := retrieval.NewRecoverer(engine, b.storeReader(), b.logger)
         b.recoveres[key] = rec
         go rec.Probe(context.Background(), ws)  // background rebuild if mismatched
       }
