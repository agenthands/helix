    **envelope.go:**
    ```go
    type Envelope struct {
        Source         Source         `json:"source"`
        FallbackReason FallbackReason `json:"fallback_reason,omitempty"`
        GraphVersion   uint64         `json:"graph_version,omitempty"`
        Freshness      string         `json:"freshness,omitempty"` // closed enum, mirrors Phase 64
        Payload        any            `json:"-"`                   // tool-specific; embedded by caller via json.Marshal merge or wrapper
    }
    func MarshalEnvelope(env Envelope, payload map[string]any) ([]byte, error) {
        // Build a flat JSON: header fields + payload kvs at the top level.
        // 65-05 (repomap) uses payload={"tree": "..."}, 65-06 (blast) uses payload={"impacts": [...]}, etc.
        out := map[string]any{ "source": env.Source }
        if env.FallbackReason != "" { out["fallback_reason"] = env.FallbackReason }
        if env.GraphVersion != 0    { out["graph_version"] = env.GraphVersion }
        if env.Freshness != ""      { out["freshness"] = env.Freshness }
        for k, v := range payload  { out[k] = v }
        return json.Marshal(out)
    }
    ```

    **source_select.go:**
    ```go
    type ConfigGate interface { SemanticIndexEnabled() bool }

    func ChooseSource(cfg ConfigGate, lookup SemanticLookup, err error) (Source, FallbackReason) {
        if cfg == nil || !cfg.SemanticIndexEnabled() {
            return SourceTreeSitter, "" // D-04 + Pitfall §3
        }
        if lookup == nil || !lookup.Available() {
            return SourceFallback, FallbackReasonIndexDisabled // defensive (D-05)
        }
        if err != nil {
            return SourceFallback, ClassifyLookupErr(err)
        }
        return SourceSemantic, ""
    }
    ```

    Pass a tiny test-side fake ConfigGate in the test (`type fakeCfg struct{ enabled bool }; func (f fakeCfg) SemanticIndexEnabled() bool { return f.enabled }`). Production-side, the daemon adapter implements ConfigGate.

    **WR-NEW-01 doctrine:** ChooseSource never reads `err.Error()`. ClassifyLookupErr (from 65-03) is the ONLY classifier; it uses errors.Is exclusively.
