    **strangler.go (NEW):**
    ```go
    package repomap

    import (
        "context"
        "github.com/agenthands/helix/internal/repomap"
        "github.com/agenthands/helix/internal/semantic/integ"
    )

    // adaptRankedFiles converts integ-shape ranked files to the renderer's
    // expected v1.9 shape. Engine source untouched per INTEG-01.
    func adaptRankedFiles(ranked []integ.RankedFile) []repomap.RankedFile {
        out := make([]repomap.RankedFile, 0, len(ranked))
        // Sort-before-iterate per Phase 62 CR-03 (S4) — sort by score desc, tiebreak by Path asc.
        sort.SliceStable(ranked, func(i, j int) bool {
            if ranked[i].Score != ranked[j].Score { return ranked[i].Score > ranked[j].Score }
            return ranked[i].Path < ranked[j].Path
        })
        for _, r := range ranked {
            out = append(out, repomap.RankedFile{File: r.Path, Score: r.Score})
        }
        return out
    }

    func computeFreshness(lookup integ.SemanticLookup, ctx context.Context, ws workspace.WorkspaceKey) string {
        if lookup == nil || !lookup.Available() { return "" }
        st, err := lookup.Status(ctx, ws)
        if err != nil { return "" }
        if st.OverlayActive && st.PendingLSP > 0 { return "structurally_fresh_semantically_pending" }
        if st.OverlayActive { return "overlay" }
        if st.PendingLSP > 0 { return "pending_lsp" }
        return "fresh"
    }
    ```
