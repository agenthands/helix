    **blast_radius_strangler.go (NEW):** Per RESEARCH Code Example 2.
    ```go
    package symbols

    import (
        "context"
        "github.com/agenthands/helix/internal/semantic/integ"
    )

    // analyzeWithLookup implements the two-pass D-07 orchestrator.
    // Pre: lookup.Available() returned true at the call site.
    func analyzeWithLookup(ctx context.Context, lookup integ.SemanticLookup, ws workspace.WorkspaceKey,
        sym integ.SymbolID, lease *Lease, uri string, line, col uint32) ([]integ.Impact, integ.Source, integ.FallbackReason, error) {

        // Pass 1: graph expansion (cheap, persisted).
        impacts, err := lookup.ExpandFrom(ctx, ws, sym, /*depth=*/ 2)
        if err != nil {
            return nil, integ.SourceFallback, integ.ClassifyLookupErr(err), err
        }

        // Pitfall §6: COPY before mutating with Pass 2 verdicts.
        impactsCopy := make([]integ.Impact, len(impacts))
        copy(impactsCopy, impacts)

        // Pass 2: filter critical edges, validate via LSP.
        critical := filterCritical(impactsCopy) // crossesPublicAPI || confidence < 0.80
        if len(critical) == 0 {
            return impactsCopy, integ.SourceSemantic, "", nil
        }
        validated, vErr := lookup.ValidateCriticalEdges(ctx, ws, edgesOf(critical))
        if vErr == nil {
            applyValidationVerdicts(impactsCopy, validated) // 1.00 confirmed / 0.20 refuted+refuted:true
        }
        // vErr non-nil: leave Pass 1 confidences; LSP unavailable is non-fatal (research Code Example 2).
        return impactsCopy, integ.SourceSemantic, "", nil
    }

    func filterCritical(impacts []integ.Impact) []integ.Impact {
        out := make([]integ.Impact, 0, len(impacts))
        for _, im := range impacts {
            if im.Confidence < 0.80 || crossesPublicAPI(im) {
                out = append(out, im)
            }
        }
        return out
    }

    func crossesPublicAPI(im integ.Impact) bool {
        // Heuristic per D-07: edge whose target is exported (capitalized) AND
        // package differs from source. Approximate via the per-node SymbolID
        // structure; full implementation reads symbol metadata from evidence.
        for _, e := range im.Evidence.Edges {
            if e.From.Package() != e.To.Package() && isExported(e.To) {
                return true
            }
        }
        return false
    }

    func applyValidationVerdicts(impacts []integ.Impact, verdicts []integ.ValidatedEdge) {
        verdict := make(map[integ.Edge]integ.ValidatedEdge, len(verdicts))
        for _, v := range verdicts { verdict[v.Edge] = v }
        for i := range impacts {
            for j := range impacts[i].Evidence.Edges {
                e := impacts[i].Evidence.Edges[j]
                if v, ok := verdict[e]; ok {
                    if v.LSPConfirmed {
                        impacts[i].Confidence = 1.00
                    } else {
                        impacts[i].Confidence = 0.20
                        impacts[i].Refuted = true
                    }
                }
            }
        }
    }
    ```
