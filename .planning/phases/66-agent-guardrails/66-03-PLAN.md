```go
type Action int
const (
    Allow Action = iota
    Warn
    Block
)

type RequiredReceipt struct {
    Class      guardrails.ReceiptClass  // closed enum from Plan 01
    ScopeHint  string                   // human-readable: "find_references on symbol X"
}

type SeeAlsoRef struct {
    Tool string
    Args map[string]string
}

type Decision struct {
    Action            Action
    Rule              string  // "G-001".."G-005" or "" for Allow
    Message           string
    RequiredReceipts  []RequiredReceipt
    SuggestedTools    []string
    SeeAlso           []SeeAlsoRef
    Warnings          []string  // populated when Action==Warn
}

// SessionContext is the predicate input bundle.
type SessionContext struct {
    Workspace        workspace.WorkspaceKey
    Profile          string
    Lookup           integ.SemanticLookup        // from Plan 02 extension
    Store            *guardrails.Store           // from Plan 01
    Config           semantic.GuardrailsConfig   // from Plan 02
    OutlineProvider  OutlineProvider             // tree-sitter symbol outline lookup (interface; production wiring in Plan 04)
    GraphVersion     uint64                      // current
    Now              time.Time
}

// OutlineProvider returns symbol names declared in a file (Go/TS/JS/Python via tree-sitter).
type OutlineProvider interface {
    SymbolsInFile(ctx context.Context, ws workspace.WorkspaceKey, path string) ([]OutlineSymbol, error)
}

type OutlineSymbol struct {
    Name        string
    Kind        string  // "function", "method", "class", "type", ...
    StartLine   int
    EndLine     int
    Visibility  integ.Visibility  // from Plan 02
}

// RuleArgs captures the per-tool argument shape that predicates inspect (NOT the receipts list — the middleware extracts that separately).
type RuleArgs struct {
    Tool          string
    Path          string
    Find          string         // fuzzy_edit/replace_in_file "find" arg
    Replace       string
    SymbolID      integ.SymbolID // resolved by middleware before predicate dispatch (or "" if unknown)
    SymbolName    string
    NewName       string
    NewBody       string
    SearchBody    string
    ChangedLines  int
    TouchedFiles  []string
    Receipts      []guardrails.ReceiptID
}
```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Agent tool args → predicate | Untrusted "find" / "path" / SymbolID / changed_lines crosses into rule logic |
| Catalog YAMLs → G-005 classifier | Operator-controlled patterns drive security classification |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-66-12 | Spoofing | Agent passes a non-identifier "find" to bypass G-001 | mitigate | regex `^[A-Za-z_][A-Za-z0-9_]{2,}$` enforced in g001 predicate; substring-of-known-symbol check is the second gate; bypass via non-match means the agent took an arguably safer path |
| T-66-13 | Tampering | Agent claims target.SymbolID is a different (harmless) symbol | mitigate | symbol-id resolution happens server-side before predicate dispatch (handled by middleware in Plan 04); plan 03 trusts only the resolved value |
| T-66-14 | Information Disclosure | G-005 catalog exposes monitored package names | accept | catalogs are operator-visible by design (D-22); no security-by-obscurity |
| T-66-15 | Denial of Service | G-004 ratio trigger inflates evaluation cost on huge files | mitigate | file_line_count is computed once and cached per (path, mtime) by the OutlineProvider in Plan 04; plan 03 predicate accepts file_line_count as input, not the file body |
| T-66-16 | Elevation of Privilege | G-003 misclassifies a private symbol as public, blocking legitimate edit | accept | Visibility=Unknown defaults to non-public-like; only Visibility=Public/Exported/Protected triggers G-003; documented in GUARDRAILS.md (Plan 06) |
| T-66-17 | Tampering | Catalog YAML embedded but operator override silently ignored | mitigate | Plan 02 already declared `import_patterns map[string][]string` in G005Config; Plan 03 catalogs.Load merges embedded defaults with cfg overrides via map-union (override wins per language) |
