// isExported checks if the last component of a stable_key starts with an uppercase letter.
// stable_key format: "pkg:TypeName:MethodName" — last ':'-delimited segment's first byte.
func isExported(stableKey string) bool {
    if i := strings.LastIndexByte(stableKey, ':'); i >= 0 && i+1 < len(stableKey) {
        return stableKey[i+1] >= 'A' && stableKey[i+1] <= 'Z'
    }
    return stableKey != "" && stableKey[0] >= 'A' && stableKey[0] <= 'Z'
}
